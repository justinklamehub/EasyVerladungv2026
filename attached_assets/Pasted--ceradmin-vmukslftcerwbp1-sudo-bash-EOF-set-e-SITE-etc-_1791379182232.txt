const fs = require("node:fs");

const file = "/etc/nginx/conf.d/easy-verladung-test.conf";
const backup = file + ".before-open-access";
const source = fs.readFileSync(file, "utf8");

const restrictedAccess = [
  "    allow 172.16.193.202;",
  "    allow 10.106.4.13;",
  "    allow 127.0.0.1;",
  "    deny all;",
  "    error_page 403 =301 https://$host$request_uri;"
].join("\n");

if (
  !source.includes("listen 80;") ||
  !source.includes("server_name 10.106.4.13;") ||
  !source.includes("root /opt/easy-verladung/app/artifacts/comet-lkw/dist/public;") ||
  source.split(restrictedAccess).length !== 2
) {
  throw new Error(
    "Stopp: Unerwartete Verladungs-Konfiguration. Nichts ändern."
  );
}

const updated = source.replace(
  restrictedAccess,
  "    # HTTP-Testzugang für alle Geräte, die diese private IP erreichen"
);

fs.copyFileSync(file, backup, fs.constants.COPYFILE_EXCL);
fs.chmodSync(backup, 0o600);
fs.writeFileSync(file, updated);

console.log("Nur die IP-Zugriffsliste der Verladungs-App entfernt.");
NODE

chmod 0644 "$SITE"
