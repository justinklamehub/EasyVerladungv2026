[ceradmin@vmukslftcerwbp1 ~]$ sudo bash <<'EOF'
set -e

UNIT="/etc/systemd/system/easy-verladung.service"
API="/opt/easy-verladung/app/artifacts/api-server"

if systemctl cat easy-verladung.service >/dev/null 2>&1 ||
   [ -e "$UNIT" ] || [ -L "$UNIT" ]; then
  echo "Stopp: Der Dienst existiert bereits."
  exit 1
fi

if ss -H -ltn 'sport = :8082' | grep -q .; then
  echo "Stopp: Port 8082 ist bereits belegt."
  exit 1
fi

cd "$API"

# Einstellungen prüfen, ohne Zugangsdaten auszugeben
runuser -u easyverladung -- /bin/node \
  --env-file=/etc/easy-verladung/server.env \
  -e '
const expected = {
  NODE_ENV: "production",
  HOST: "127.0.0.1",
  PORT: "8082",
  DISABLE_SCHEDULER: "true",
  COOKIE_SECURE: "true"
};

for (const [key, value] of Object.entries(expected)) {
  if (process.env[key] !== value) {
    console.error("Stopp: Einstellung prüfen: " + key);
    process.exit(1);
  }
}

for (const key of ["DATABASE_URL", "SESSION_SECRET"]) {
  if (!process.env[key]) {
    console.error("Stopp: Einstellung fehlt: " + key);
    process.exit(1);
  }
}

console.log("Sicherheitswerte geprüft.");
'

cat > "$UNIT" <<'UNITFILE'

Description=Easy Verladung API
